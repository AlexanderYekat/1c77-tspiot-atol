функция АдресСервисаПечатиЧека(Путь, ПрефиксМаршрута = "/kkt")
	хост = СокрЛП(глЭтотМодульХостСервисаПечатиЧека);
	порт = СокрЛП(глЭтотМодульПортСервисаПечатиЧек);
	Если хост = "" тогда
		хост = "localhost";
	конецЕсли;
	Если порт = "" тогда
		порт = "2579";
	конецЕсли;
	база = "http://" + хост + ":" + порт;
	Если СокрЛП(ПрефиксМаршрута) <> "" тогда
		база = база + ПрефиксМаршрута;
	конецЕсли;
	Если СокрЛП(Путь) = "" тогда
		Возврат база;
	конецЕсли;
	Если Лев(СокрЛП(Путь), 1) <> "/" тогда
		Путь = "/" + Путь;
	конецЕсли;
	Возврат база + Путь;
конецФункции //АдресСервисаПечатиЧека

функция ПолучитьЗначениеПоляJSONПоКлючу(json, ключ, ключ2 = "")
	ЗначКлюча = "";
	паттерн = """" + ключ + """:";
	начПозНазвКлюч = Найти(json, паттерн);
	Если начПозНазвКлюч > 0 тогда
		СтрОбрабЛок = Сред(json, начПозНазвКлюч + СтрДлина(паттерн));
		Если ключ2 <> "" тогда
			Возврат ПолучитьЗначениеПоляJSONПоКлючу(СтрОбрабЛок, ключ2);
		иначе
			конецЗнач = Найти(СтрОбрабЛок, ",");
			Если конецЗнач = 0 Тогда
				конецЗнач = Найти(СтрОбрабЛок, "}");
			КонецЕсли;
			Если (конецЗнач > 0) и (Сред(СтрОбрабЛок, конецЗнач, 1) = ",") тогда
				Если Сред(СтрОбрабЛок, конецЗнач - 1, 1) = "}" тогда
					конецЗнач = конецЗнач - 1;
				конецЕсли;
			конецЕсли;
			Если конецЗнач > 0 тогда
				СтрОбрабЛок = СокрЛП(Лев(СтрОбрабЛок, конецЗнач - 1));
			конецЕсли;
			ЗначКлюча = СокрЛП(СтрЗаменить(СтрОбрабЛок, "\""", ""));
			ЗначКлюча = СокрЛП(СтрЗаменить(ЗначКлюча, """", ""));
		конецЕсли;
	конецЕсли;
	Возврат ЗначКлюча;
конецФункции //ПолучитьЗначениеПоляJSONПоКлючу

		// на розн.складе всем разрашаем выбирать цену из списка
	    Форма.Цена.Доступность(1);
		Форма.Цена.Редактирование(0);
	КонецЕсли;
	
	// если дата проведенного документа больше ТА, то открываем только на просмотр,
	// так как его все равно не удастся сохранить после редактирования.
	Если (Проведен() = 1) И (ДатаДок > ПолучитьДатуТА()) Тогда
		Форма.ТолькоПросмотр(1);
	КонецЕсли;
	
	//Инициализирум список действий по кнопке "Действия"
	СписокДействий = СоздатьОбъект("СписокЗначений");
	СписокДействий.ДобавитьЗначение("Отчет о движениях документа");
	СписокДействий.ДобавитьЗначение("Структура подчиненности");
              
	// Если открыли только на просмотр, то надо кнопки сделать недоступными
	Если Форма.ТолькоПросмотр()=1 Тогда
	    // СДЕЛАТЬ НЕДОСТУПНЫМИ КНОПКИ МОДИФИКАЦИИ ДОКУМЕНТА
		Форма.кнОК.Доступность(0);
		Форма.кнЗаписать.Доступность(0);
		Форма.кнПровести.Доступность(0);
		Форма.кнПодбор.Доступность(0);
	Иначе	
		СписокДействий.ДобавитьЗначение("Добавить из документа");
	КонецЕсли;
	
	СписокДействий.ДобавитьЗначение("Ввести на основании");
	СписокДействий.ДобавитьЗначение("Перейти в журнал");
	
	// кнопка по умолчанию
	Если (Форма.ТолькоПросмотр() = 1) ИЛИ ((ДатаДок < Макс(РабочаяДата(), ПолучитьДатуТА()) ) И (Выбран() = 1)) Тогда
	    Форма.КнопкаПоУмолчанию("кнЗакрыть");
	Иначе
	    Форма.КнопкаПоУмолчанию("кнОК");
	КонецЕсли;
	
	ОбновитьНадписи();    
	УправлениеВидимостью(); 
	
	//изм  
	Если глПолучитьПолномочие("РазрешитьРедактированиеЦенВдокументах") = 0 Тогда
	Форма.кнПодбор.Видимость(0);
	Форма.кнПодбор.Доступность(0); 
	КонецЕсли;
	
	глДПВключитьРежимВыводаСтрок();
	
	//Если документ еще не проведен, тогда 
	//проведение делаем только в потоке
	Если ( Проведен() = 0 ) Тогда
		ПроводитьПослеТА(1,1);
	КонецЕсли;  
	
	Если (НазваниеНабораПрав() = "АдминистраторБезУдаления") ИЛИ (НазваниеНабораПрав() = "Администратор") Тогда
	Иначе
		Форма.Скидка.Доступность(0);
//		Форма.Клиент.Доступность(0);
		Форма.кнНетСкидки.Доступность(0);
		Форма.кнДокумент.Доступность(0);
		Форма.кнДокумент.Видимость(0);
	КонецЕсли;

	//август 2026. ТСПиОТ — init внешнего сервиса печати/проверки марок
	Если глБылЗапущенМодульМоудльФормыЧека = 0 тогда
		Сообщить("Глобальный модуль версии " + СокрЛП(глВерсияГлобальногоМодуля));		
		Сообщить("Модуль """ + СокрЛП(глЭтотМодульИмяМодкуля) + """ версии " + СокрЛП(глЭтотМодульВерсияМодуля));
		глБылЗапущенМодульМоудльФормыЧека = 1;
	конецЕсли;				

	глЭтотМодульХостСервисаПечатиЧека = "localhost";
	глЭтотМодульПортСервисаПечатиЧек = "2579";
	Если ЧекПробитККМ = 0 тогда
		резОш = 0;
		резОписОшибки = "";
		глЭтотМодульВнешнийСервисПечатиЧекаДоступен = ПроверитьДоступностьСервисаПечатиЧека(глЭтотМодульХостСервисаПечатиЧека, глЭтотМодульПортСервисаПечатиЧек, резОш, резОписОшибки);
		глСервисПечатиЧекаДоступен = глЭтотМодульВнешнийСервисПечатиЧекаДоступен;
		глХостСервисаПечатиЧека = СокрЛП(глЭтотМодульХостСервисаПечатиЧека);
		глПортСервисаПечатиЧека = СокрЛП(глЭтотМодульПортСервисаПечатиЧек);
		Если (резОш = 0) и (глЭтотМодульВнешнийСервисПечатиЧекаДоступен = 1) тогда
			Сообщить("бьём чеки через сервис программу");
			ОчиститьБуферПроверенныхМарокБезОжидания();
		иначе
			Сообщить("Печать чеков напрямую через комнады ККТ!");
		конецЕсли;
	конецЕсли;
КонецПроцедуры // ПриОткрытии()
