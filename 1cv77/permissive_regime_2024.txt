// Низкоуровневый HTTP через WinHttp.WinHttpRequest.5.1
// Возвращает список: body, statusCode, statusText
функция HTTPОтправитьЗапросWinHttp(Метод, Адрес, JSONТело, кодОшибки, ОписаниеРезультата = "", ВыводитьОписаниеОшибок = 0)
	спРез = СоздатьОбъект("СписокЗначений");
	спРез.Установить("body", "");
	спРез.Установить("statusCode", 0);
	спРез.Установить("statusText", "");
	кодОшибки = 0;
	ОписаниеРезультата = "";
	Метод = ВРег(СокрЛП(Метод));
	если Метод = "" тогда
		Метод = "POST";
	конецЕсли;
	если (Метод = "POST") и (СокрЛП(JSONТело) = "") тогда
		JSONТело = "{}";
	конецЕсли;
	try
		HTTP = СоздатьОбъект("WinHttp.WinHttpRequest.5.1");
		таймаут = глHTTPТаймаутМс;
		если таймаут <= 0 тогда
			таймаут = 30000;
		конецЕсли;
		HTTP.SetTimeouts(таймаут, таймаут, таймаут, таймаут);
		HTTP.Open(Метод, Адрес, 0);
		HTTP.SetRequestHeader("Content-Type", "application/json");
		HTTP.SetRequestHeader("Accept", "application/json");
		если ВыводитьОписаниеОшибок = 1 тогда
			Сообщить(Метод + " " + Адрес + " [WinHttp, таймаут=" + СокрЛП(таймаут) + " мс]");
		конецЕсли;
		если Метод = "GET" тогда
			HTTP.Send();
		иначе
			//HTTP.Send(ПреобразоватьСтрокуВУТФ8(JSONТело));
			HTTP.Send(JSONТело);
		конецЕсли;
		спРез.Установить("statusCode", HTTP.Status);
		спРез.Установить("statusText", СокрЛП(HTTP.StatusText));
		//спРез.Установить("body", ПреобразоватьСтрокуИзУТФ8(HTTP.ResponseText));
		спРез.Установить("body", HTTP.ResponseText);
	except
		ОписаниеРезультата = СокрЛП(ОписаниеОшибки());
		если ВыводитьОписаниеОшибок = 1 тогда
			Сообщить(ОписаниеРезультата, "!");
		конецЕсли;
		кодОшибки = 10;
	endtry;
	возврат спРез;
конецФункции //HTTPОтправитьЗапросWinHttp

// Низкоуровневый HTTP через AddIn.V7HttpReader (глСоединение должен быть подключён)
// Возвращает список: body, statusCode, statusText
функция HTTPОтправитьЗапросV7Plus(Метод, Адрес, JSONТело, кодОшибки, ОписаниеРезультата = "", ВыводитьОписаниеОшибок = 0)
	спРез = СоздатьОбъект("СписокЗначений");
	спРез.Установить("body", "");
	спРез.Установить("statusCode", 0);
	спРез.Установить("statusText", "");
	кодОшибки = 0;
	ОписаниеРезультата = "";
	Метод = ВРег(СокрЛП(Метод));
	если Метод = "" тогда
		Метод = "POST";
	конецЕсли;
	если (Метод = "POST") и (СокрЛП(JSONТело) = "") тогда
		JSONТело = "{}";
	конецЕсли;
	стрПол = "";
	try
		глСоединение.УдалитьЗаголовкиЗапроса();
		глСоединение.УстановитьЗаголовокЗапроса("Content-Type", "application/json");
		глСоединение.УстановитьЗаголовокЗапроса("Accept", "application/json");
		если ВыводитьОписаниеОшибок = 1 тогда
			Сообщить(Метод + " " + Адрес + " [V7Plus]");
		конецЕсли;
		если Метод = "GET" тогда
			глСоединение.Получить(Адрес, стрПол, 2);
		иначе
			глСоединение.ОтправитьДляОбработки(Адрес, ПреобразоватьСтрокуВУТФ8(JSONТело), 2, стрПол, 2);
		конецЕсли;
	except
		ОписаниеРезультата = СокрЛП(ОписаниеОшибки());
		если ВыводитьОписаниеОшибок = 1 тогда
			Сообщить(ОписаниеРезультата, "!");
		конецЕсли;
		кодОшибки = 10;
		возврат спРез;
	endtry;
	стрСостояния = "";
	try
		стрСостояния = глСоединение.СтрокаСостоянияОтвета;
	except
	endtry;
	спРез.Установить("statusCode", HTTPКодСостоянияИзОтвета(стрСостояния));
	спРез.Установить("statusText", стрСостояния);
	спРез.Установить("body", ПреобразоватьСтрокуИзУТФ8(стрПол));
	возврат спРез;
конецФункции //HTTPОтправитьЗапросV7Plus

// Fire-and-forget: sync Send + короткие таймауты; ответ/ошибку не проверяем.
// Sync нужен, чтобы POST успел уйти до уничтожения COM-объекта HTTP
// (при Open(..., 1) async объект часто уничтожается раньше, чем запрос дойдёт до сервера).
// Проверка доставки: kktserver.log — строки ClearBufferOfMarksJson begin / OK.
процедура HTTPОтправитьЗапросWinHttpБезОжидания(Метод, Адрес, JSONТело)
	Метод = ВРег(СокрЛП(Метод));
	если Метод = "" тогда
		Метод = "POST";
	конецЕсли;
	если (Метод = "POST") и (СокрЛП(JSONТело) = "") тогда
		JSONТело = "{}";
	конецЕсли;
	таймаут = глHTTPТаймаутБезОжиданияМс;
	если таймаут <= 0 тогда
		таймаут = 500;
	конецЕсли;
	try
		HTTP = СоздатьОбъект("WinHttp.WinHttpRequest.5.1");
		// resolve, connect, send — коротко; receive — минимально (ответ не важен)
		HTTP.SetTimeouts(таймаут, таймаут, таймаут, 100);
		HTTP.Open(Метод, Адрес, 0);
		HTTP.SetRequestHeader("Content-Type", "application/json");
		HTTP.SetRequestHeader("Accept", "application/json");
		если Метод = "GET" тогда
			HTTP.Send();
		иначе
			HTTP.Send(JSONТело);
		конецЕсли;
	except
		// Сервер недоступен или таймаут — не блокируем вызывающий код
	endtry;
конецПроцедуры //HTTPОтправитьЗапросWinHttpБезОжидания

// Извлекает объекты {...} из массива JSON по имени, напр. "marks"
функция ИзвлечьОбъектыИзМассиваJSON(json, имяМассива)
	сп = СоздатьОбъект("СписокЗначений");
	ключ = """" + имяМассива + """:";
	позКлюч = Найти(json, ключ);
	если позКлюч = 0 тогда
		возврат сп;
	конецЕсли;
	поз = позКлюч + СтрДлина(ключ);
	дл = СтрДлина(json);
	пока (поз <= дл) и (Найти(" " + Симв(9) + Симв(10) + Симв(13), Сред(json, поз, 1)) > 0) цикл
		поз = поз + 1;
	конецЦикла;
	если (поз > дл) или (Сред(json, поз, 1) <> "[") тогда
		возврат сп;
	конецЕсли;
	поз = поз + 1;
	вКавычках = 0;
	экран = 0;
	уровеньФиг = 0;
	начЭлем = 0;
	пока поз <= дл цикл
		с = Сред(json, поз, 1);
		если экран = 1 тогда
			экран = 0;
		иначеесли с = "\" тогда
			экран = 1;
		иначеесли (вКавычках = 0) и (с = "]") и (уровеньФиг = 0) тогда
			прервать;
		иначеесли с = """" тогда
			вКавычках = ?(вКавычках = 1, 0, 1);
		иначеесли вКавычках = 0 тогда
			если с = "{" тогда
				если уровеньФиг = 0 тогда

				statusUp = НРег(СокрЛП(спРазбор.Получить("status")));
				delphiUp = НРег(СокрЛП(спРазбор.Получить("delphiUp")));
				proxyOk = 0;
				если (proxyUp = "true") или (proxyUp = "1") или (statusUp = "up") тогда

						delphiOk = 1;
					конецЕсли;

					доступен = 1;
				конецЕсли;
			конецЕсли;
		конецЕсли;
		спПараметры.Установить("доступен", доступен);
		Если доступен = 1 тогда
			спПараметры.Установить("ipХоста", выбIpСервер);

			спПараметры.Установить("описНепринятыхРР", спРазбор.Получить("описНепринятыхРР"));
		конецЕсли;

    ЦДО=СоздатьОбъект("CDO.Message");
    БодиПарт=ЦДО.BodyPart;
    Поля=БодиПарт.Fields;
    Поле=Поля.Item("urn:schemas:mailheader:content-type");
    Поле.Value="text/plain; charset=""windows-1251""";
    Поля.Update();
    Стрим = БодиПарт.GetDecodedContentStream();
    Стрим.charset = "windows-1251";
    Стрим.WriteText(Стр);
    Стрим.Flush();
    БодиПарт.ContentTransferEncoding = Кодировка;
    Стрим=БодиПарт.GetEncodedContentStream();
    Возврат Стрим.ReadText();

Функция ДекодироватьBase64(Текст64)
    Перем Алфавит;
    Перем Группа, Символ;
    Перем Значение, Разряд;
    Перем Символ6, Значение6;
    Перем Значение8, Символ8;
    Перем Текст;
    
    // Подготовка алфавита и результата
    Алфавит = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/=";
    Текст = "";
    
    // Получение групп по 4 значащих символа
    Группа = "";
    Для Номер = 1 По СтрДлина(Текст64) Цикл
        Символ = Сред(Текст64, Номер, 1);
        Если Найти(Алфавит, Символ) = 0 Тогда
            Продолжить;
        КонецЕсли;
        Группа = Группа + Символ;
        Если СтрДлина(Группа) = 4 Тогда


            // Получение числового значения из 4-х символов
            Значение = 0;
            Разряд = 3;
            Пока Разряд >= 0 Цикл
                Символ6 = Сред(Группа, 4 - Разряд, 1);
                Если Символ6 = "=" Тогда
                    Значение6 = 0;
                Иначе
                    Значение6 = Найти(Алфавит, Символ6) - 1;
                КонецЕсли;
                Если Разряд = 1 Тогда
                    Значение6 = Значение6 * 64;
                ИначеЕсли Разряд = 2 Тогда
                    Значение6 = Значение6 * 64 * 64;
                ИначеЕсли Разряд = 3 Тогда
                    Значение6 = Значение6 * 64 * 64 * 64;
                КонецЕсли;
                Значение = Значение + Значение6;
                Разряд = Разряд - 1;
            КонецЦикла;


            // Получение 3-х символов из числового значения            
            Разряд = 2;
            Пока Разряд >= 0 Цикл
                Значение8 = Значение;
                Если Разряд = 2 Тогда
                    Значение8 = Цел(Значение8 / 256 / 256);
                    Значение = Значение - Значение8 * 256 * 256;
                ИначеЕсли Разряд = 1 Тогда
                    Значение8 = Цел(Значение8 / 256);
                    Значение = Значение - Значение8 * 256;
                КонецЕсли;
                
                Если Значение8 <>  0 Тогда // [+] Decker
                    // чтобы не добавлялись лишние символы пробела
                    Символ8 = Симв(Значение8);
                    Текст = Текст + Символ8;
                КонецЕсли;
                
                Разряд = Разряд - 1;
            КонецЦикла;
            
            Группа = "";
        КонецЕсли;
    КонецЦикла;
    
    Возврат Текст;
    
КонецФункции // ДекодироватьBase64()

    Перем Символ6, Значение6;
    Перем Значение8, Символ8;
    Перем Текст64;
    
    // Подготовка алфавита и результата
    Алфавит = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/=";
    Текст64 = ""; 
           
    Группа = "";
    Для Номер=1 По СтрДлина(Текст) Цикл
    
    Символ = Сред(Текст, Номер, 1);
    Группа = Группа + Символ;
    
    Если ((СтрДлина(Группа) = 3) или (Номер = СтрДлина(Текст))) Тогда
        
        // Получение числового значения из 3-х символов
            Значение = 0;
            Разряд = 2;
            Пока Разряд >= 0 Цикл
                Символ8 = Сред(Группа, 3 - Разряд, 1);
                Значение8 = КодСимв(Символ8);
                Если Разряд = 1 Тогда
                    Значение8 = Значение8 * 256;
                ИначеЕсли Разряд = 2 Тогда
                    Значение8 = Значение8 * 256 * 256;
                КонецЕсли;
                Значение = Значение + Значение8;
                Разряд = Разряд - 1;
            КонецЦикла;
            
        // Получение 4-х символов из числового значения    
            Разряд = 3;
            Пока Разряд >= 0 Цикл
                Значение6 = Значение;
                Если Разряд = 3 Тогда
                    Значение6 = Цел(Значение6 / 64 / 64 / 64);
                    Значение = Значение - Значение6 * 64 * 64 * 64;
                ИначеЕсли Разряд = 2 Тогда
                    Значение6 = Цел(Значение6 / 64 / 64);
                    Значение = Значение - Значение6 * 64 * 64;
                ИначеЕсли Разряд = 1 Тогда
                    Значение6 = Цел(Значение6 / 64);
                    Значение = Значение - Значение6 * 64;
                КонецЕсли;      
                
                Если СтрДлина(Группа) < 3 Тогда
                    Если Значение6 = 0 Тогда
                        Значение6 = 64;
                    КонецЕсли;
                КонецЕсли;


                Символ6 = Сред(Алфавит, Значение6+1,1);
                Текст64 = Текст64 + Символ6;
                Разряд = Разряд - 1;
            КонецЦикла;    
                        
        Группа = "";    
        КонецЕсли;
    КонецЦикла;   
    
    Возврат Текст64;

Функция ЕстьПолныйФормат(Маркировка)
    // Длина полного формата без разделителей:
    // 01 + 14 цифр (первая группа) = 16 символов
    // 21 + 13 символов (вторая группа) = 15 символов
    // 91 + 4 символа (третья группа) = 6 символов
    // 92 + 44 символа (четвертая группа) = 46 символов
    // Итого минимум 83 символа для полного формата
    
    // Длина сокращенного формата без разделителей:
    // 01 + 14 цифр (первая группа) = 16 символов
    // 21 + 13 символов (вторая группа) = 15 символов
    // Итого 31 символ для сокращенного формата
    
    ДлинаСтроки = СтрДлина(Маркировка);
    
    // Учитываем возможные разделители (максимум 3 для полного формата)
    // Если длина больше 40, считаем что это полный формат
    Возврат ?(ДлинаСтроки > 40, 1, 0);
