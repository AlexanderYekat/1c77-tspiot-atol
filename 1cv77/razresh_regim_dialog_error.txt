функция ГексВЧисло(гекс)
	гекс = Врег(СокрЛП(гекс));
	цифры = "0123456789ABCDEF";
	рез = 0;
	н = 1;

		п = Найти(цифры, Сред(гекс, н, 1));
		Если п = 0 тогда
			возврат -1;
		конецЕсли;
		рез = рез * 16 + (п - 1);

	конецЦикла;

конецФункции //ГексВЧисло

// Unicode codepoint -> символ Windows-1251 (1С 7.7)
функция ЮникодКодВСимвол(код)
	Если (код >= 0) и (код <= 127) тогда
		возврат Симв(код);
	конецЕсли;
	// Ё U+0401, ё U+0451
	Если код = 1025 тогда
		возврат Симв(168);
	конецЕсли;
	Если код = 1105 тогда
		возврат Симв(184);
	конецЕсли;
	// А..я U+0410..U+044F -> CP1251 0xC0..0xFF
	Если (код >= 1040) и (код <= 1103) тогда
		возврат Симв(код - 848);
	конецЕсли;
	Если (код >= 0) и (код <= 255) тогда
		возврат Симв(код);
	конецЕсли;
	возврат "";
конецФункции //ЮникодКодВСимвол

// Как UnescapeJsonUnicode в службе: \u041C\u0430... -> "Ма..."
функция РаскодироватьUnicodeEscapes(стр)
	стр = "" + стр;
	Если (Найти(стр, "\u") = 0) и (Найти(стр, "\U") = 0) тогда
		возврат стр;
	конецЕсли;
	рез = "";
	дл = СтрДлина(стр);
	н = 1;
	Пока н <= дл цикл
		добавлен = 0;
		Если (н + 5 <= дл) и (Сред(стр, н, 1) = "\") и (Врег(Сред(стр, н + 1, 1)) = "U") тогда
			код = ГексВЧисло(Сред(стр, н + 2, 4));
			Если код >= 0 тогда
				сим = ЮникодКодВСимвол(код);
				Если сим <> "" тогда
					рез = рез + сим;
					н = н + 6;
					добавлен = 1;
				конецЕсли;
			конецЕсли;
		конецЕсли;
		Если добавлен = 0 тогда
			рез = рез + Сред(стр, н, 1);
			н = н + 1;
		конецЕсли;
	конецЦикла;
	возврат рез;
конецФункции //РаскодироватьUnicodeEscapes
