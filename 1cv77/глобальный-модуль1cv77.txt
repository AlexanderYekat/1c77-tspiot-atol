функция глПолучитьЗначениеПоляJSONПоКлючу(json, ключ, ключ2 = "")
	ЗначКлюча = "";
	паттерн = """" + ключ + """:";
	начПозНазвКлюч = Найти(json, паттерн);
	Если начПозНазвКлюч > 0 тогда
		СтрОбрабЛок = Сред(json, начПозНазвКлюч + СтрДлина(паттерн));
		Если ключ2 <> "" тогда
			Возврат глПолучитьЗначениеПоляJSONПоКлючу(СтрОбрабЛок, ключ2);
		иначе
			конецЗнач = Найти(СтрОбрабЛок, ",");
			Если конецЗнач = 0 Тогда
				конецЗнач = Найти(СтрОбрабЛок, "}");
			КонецЕсли;
			Если (конецЗнач > 0) и (Сред(СтрОбрабЛок, конецЗнач, 1) = ",") тогда
				Если Сред(СтрОбрабЛок, конецЗнач - 1, 1) = "}" тогда
					конецЗнач = конецЗнач - 1;
				конецЕсли;
			конецЕсли;
			Если конецЗнач > 0 тогда
				СтрОбрабЛок = СокрЛП(Лев(СтрОбрабЛок, конецЗнач - 1));
			конецЕсли;
			ЗначКлюча = СокрЛП(СтрЗаменить(СтрОбрабЛок, "\""", ""));
			ЗначКлюча = СокрЛП(СтрЗаменить(ЗначКлюча, """", ""));
		конецЕсли;
	конецЕсли;
	Возврат ЗначКлюча;
конецФункции //глПолучитьЗначениеПоляJSONПоКлючу

// Короткий таймаут: один GET status не должен тормозить кассу.
функция глHTTPОтправитьЗапросWinHttpКороткий(Метод, Адрес, JSONТело, кодОшибки, ОписаниеРезультата = "")
	кодОшибки = 0;
	ОписаниеРезультата = "";
	тело = "";
	Метод = ВРег(СокрЛП(Метод));
	Если Метод = "" тогда
		Метод = "GET";
	конецЕсли;
	Если (Метод = "POST") и (СокрЛП(JSONТело) = "") тогда
		JSONТело = "{}";
	конецЕсли;
	таймаут = 2000;
	Попытка
		HTTP = СоздатьОбъект("WinHttp.WinHttpRequest.5.1");
		HTTP.SetTimeouts(таймаут, таймаут, таймаут, таймаут);
		HTTP.Open(Метод, Адрес, 0);
		HTTP.SetRequestHeader("Content-Type", "application/json");
		HTTP.SetRequestHeader("Accept", "application/json");
		Если Метод = "GET" тогда
			HTTP.Send();
		иначе
			HTTP.Send(JSONТело);
		конецЕсли;
		кодHTTP = HTTP.Status;
		тело = HTTP.ResponseText;
		Если (кодHTTP <> 0) и (кодHTTP <> 200) тогда
			кодОшибки = 20;
			ОписаниеРезультата = "HTTP " + СокрЛП(кодHTTP) + ": " + СокрЛП(HTTP.StatusText);
		конецЕсли;
	Исключение
		кодОшибки = 10;
		ОписаниеРезультата = СокрЛП(ОписаниеОшибки());
	КонецПопытки;
	Возврат тело;
конецФункции //глHTTPОтправитьЗапросWinHttpКороткий

процедура глОбновитьЗаписьПоОтветуStatus(марка, спЗапись, тело, кодОш, описОш)
	taskId = СокрЛП(спЗапись.Получить("taskId"));
	Если кодОш <> 0 тогда
		глЗаписатьРезультатПроверкиМарки(марка, taskId, "ошибка", описОш, "");
		Возврат;
	конецЕсли;
	статусAPI = НРег(СокрЛП(глПолучитьЗначениеПоляJSONПоКлючу(тело, "status")));
	опис = СокрЛП(глПолучитьЗначениеПоляJSONПоКлючу(тело, "description"));
	allAccepted = СокрЛП(глПолучитьЗначениеПоляJSONПоКлючу(тело, "allAccepted"));
	allPermissionAccepted = СокрЛП(глПолучитьЗначениеПоляJSONПоКлючу(тело, "allPermissionAccepted"));
	Если статусAPI = "pending" тогда
		глЗаписатьРезультатПроверкиМарки(марка, taskId, "проверяется", опис, allAccepted);
	ИначеЕсли статусAPI = "completed" тогда
		// allAccepted = ККТ; allPermissionAccepted = РР (если checkPermission был true)
		acc = НРег(allAccepted);
		perm = НРег(allPermissionAccepted);
		кктОк = (acc = "true") или (acc = "1");
		// пустой allPermissionAccepted — РР не запрашивали, не блокируем
		ррОк = (perm = "") или (perm = "true") или (perm = "1");
		Если кктОк и ррОк тогда
			глЗаписатьРезультатПроверкиМарки(марка, taskId, "ok", опис, allAccepted);
		иначе
			описККТ = "";
			описРР = "";
			Если кктОк = 0 тогда
				описККТ = СокрЛП(глПолучитьЗначениеПоляJSONПоКлючу(тело, "marks", "description"));
				Если описККТ = "" тогда
					описККТ = опис;
				конецЕсли;
				Если (описККТ = "") или (НРег(описККТ) = "ok") тогда
					описККТ = "марка не принята ККТ";
				конецЕсли;
			конецЕсли;
			Если ррОк = 0 тогда
				описРР = СокрЛП(глПолучитьЗначениеПоляJSONПоКлючу(тело, "permission", "error"));
				Если описРР = "" тогда
					описРР = СокрЛП(глПолучитьЗначениеПоляJSONПоКлючу(тело, "error"));
				конецЕсли;
				описРР = СокрЛП(СтрЗаменить(описРР, Символ(13), " "));
				описРР = СокрЛП(СтрЗаменить(описРР, Символ(10), " "));
				Если описРР = "" тогда
					описРР = "РР не пройден";
				конецЕсли;
			конецЕсли;
			Если (описККТ <> "") и (описРР <> "") тогда
				опис = "ККТ: " + описККТ + "; РР: " + описРР;
			ИначеЕсли описРР <> "" тогда
				опис = описРР;
			ИначеЕсли описККТ <> "" тогда
				опис = описККТ;
			иначе
				опис = "марка не принята";
			конецЕсли;
			глЗаписатьРезультатПроверкиМарки(марка, taskId, "ошибка", опис, allAccepted);
		конецЕсли;
	иначе
		// error / неизвестный статус
		Если опис = "" тогда
			опис = статусAPI;
		конецЕсли;
		Если опис = "" тогда
			опис = "ошибка проверки";
		конецЕсли;
		глЗаписатьРезультатПроверкиМарки(марка, taskId, "ошибка", опис, allAccepted);
	конецЕсли;
конецПроцедуры //глОбновитьЗаписьПоОтветуStatus
