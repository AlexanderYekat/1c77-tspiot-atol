Функция Подключить(Компонента, ПрогИд, Объект, Пароль)
     
	Возврат 1;
	Рез = 0;
	Если ЗагрузитьВнешнююКомпоненту(Компонента) = 0  Тогда
		ОписаниеРезультата = "не удалось загрузить внешнюю компоненту """ + Компонента + """";
	Иначе
		Попытка
			Объект = СоздатьОбъект("AddIn." + ПрогИд);//Создание объекта внешней компоненты
		Исключение
		КонецПопытки;
		Если ПустоеЗначение(Объект) = 1 Тогда
			ОписаниеРезультата = "не удалось создать объект внешней компоненты с программым идентификатром AddIn." + ПрогИд;
		Иначе
			Объект.Password=Пароль;
			Объект.Connect();
			Ответ = Объект.ResultCode;
			Если Ответ=0 Тогда
				Рез=1;
			Иначе
				ОписаниеРезультата = ПолучитьОписаниеРезультата(Объект);
			КонецЕсли;
		КонецЕсли;
	КонецЕсли;
	Возврат Рез;

		Рез = 1;//Нет объекта, ничего не нужно отключать
	Иначе

		Объект.Password=Пароль;
		Объект.Disconnect();
		Ответ = Объект.ResultCode;
		Если Ответ=0 Тогда
			Рез=1;
		Иначе
			ОписаниеРезультата = ПолучитьОписаниеРезультата(Объект);
		КонецЕсли;
	КонецЕсли;

Функция JSONЭкранироватьСтроку(Стр)
	рез = "";
	Для н = 1 По СтрДлина(Стр) Цикл
		симвв = Сред(Стр, н, 1);
		Если симвв = "\" Тогда
			рез = рез + "\\";
		ИначеЕсли симвв = """" Тогда
			рез = рез + "\""";
		ИначеЕсли симвв = Симв(10) Тогда
			рез = рез + "\n";
		ИначеЕсли симвв = Симв(13) Тогда
			рез = рез + "\r";
		ИначеЕсли симвв = Симв(9) Тогда
			рез = рез + "\t";
		Иначе
			рез = рез + симвв;
		КонецЕсли;
	КонецЦикла;
	Возврат рез;
КонецФункции //JSONЭкранироватьСтроку

функция ПодготовитьМаркуДляККТ(МаркаСырая)
	// В реквизите Маркировка GS (Chr29) часто хранится как "0029".
	// ККТ (FNCheckItemBarcode) принимает только настоящий Симв(29).
	Марка = СокрЛП(МаркаСырая);
	Если ПустаяСтрока(Марка) = 1 Тогда
		Возврат "";
	КонецЕсли;
	Если Найти(Марка, Симв(29)) > 0 Тогда
		Возврат Марка;
	КонецЕсли;
	Если Найти(Марка, "0029") > 0 Тогда
		Возврат СтрЗаменить(Марка, "0029", Симв(29));
	КонецЕсли;
	// GS вырезаны совсем — собрать по позициям (префикс 31 + 91xxxx + 92...)
	Возврат СокрЛП(Сред(Марка, 1, 31) + Симв(29) + Сред(Марка, 32, 6) + Симв(29) + Сред(Марка, 38));
конецФункции //ПодготовитьМаркуДляККТ

Функция КодироватьBase64(Текст)
	Перем Алфавит;
	Перем Группа, Символ;
	Перем Значение, Разряд;
	Перем Символ6, Значение6;
	Перем Значение8, Символ8;
	Перем Текст64;
	Алфавит = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/=";
	Текст64 = "";
	Группа = "";
	Для Номер = 1 По СтрДлина(Текст) Цикл
		Символ = Сред(Текст, Номер, 1);
		Группа = Группа + Символ;
		Если ((СтрДлина(Группа) = 3) или (Номер = СтрДлина(Текст))) Тогда
			Значение = 0;
			Разряд = 2;
			Пока Разряд >= 0 Цикл
				Символ8 = Сред(Группа, 3 - Разряд, 1);
				Значение8 = КодСимв(Символ8);
				Если Разряд = 1 Тогда
					Значение8 = Значение8 * 256;
				ИначеЕсли Разряд = 2 Тогда
					Значение8 = Значение8 * 256 * 256;
				КонецЕсли;
				Значение = Значение + Значение8;
				Разряд = Разряд - 1;
			КонецЦикла;
			Разряд = 3;
			Пока Разряд >= 0 Цикл
				Значение6 = Значение;
				Если Разряд = 3 Тогда
					Значение6 = Цел(Значение6 / 64 / 64 / 64);
					Значение = Значение - Значение6 * 64 * 64 * 64;
				ИначеЕсли Разряд = 2 Тогда
					Значение6 = Цел(Значение6 / 64 / 64);
					Значение = Значение - Значение6 * 64 * 64;
				ИначеЕсли Разряд = 1 Тогда
					Значение6 = Цел(Значение6 / 64);
					Значение = Значение - Значение6 * 64;
				КонецЕсли;
				Если СтрДлина(Группа) < 3 Тогда
					Если Значение6 = 0 Тогда
						Значение6 = 64;
					КонецЕсли;
				КонецЕсли;
				Символ6 = Сред(Алфавит, Значение6 + 1, 1);
				Текст64 = Текст64 + Символ6;
				Разряд = Разряд - 1;
			КонецЦикла;
			Группа = "";
		КонецЕсли;
	КонецЦикла;
	Возврат Текст64;
КонецФункции //КодироватьBase64

процедура ПеренестиПринятиеМарокККТВТаблицуЧека(СписокТоваров, списокМарок)
	Если (ПустоеЗначение(СписокТоваров) = 1) или (ПустоеЗначение(списокМарок) = 1) тогда
		возврат;
	конецЕсли;
	ОбеспечитьКолонкуМаркаПринятаККТ(СписокТоваров);
	Попытка
		тестUUID = СписокТоваров.UUID;
	Исключение
		СписокТоваров.НоваяКолонка("UUID",    "строка", , , , 40);
		СписокТоваров.НоваяКолонка("time",    "строка", , , , 30);
		СписокТоваров.НоваяКолонка("Inst",    "строка", , , , 30);
		СписокТоваров.НоваяКолонка("Version", "строка", , , , 30);
	КонецПопытки;
	колМарок = списокМарок.РазмерСписка();
	СписокТоваров.ВыбратьСтроки();
	Пока СписокТоваров.ПолучитьСтроку() = 1 Цикл
		СписокТоваров.МаркаПринятаККТ = 0;
		МаркаВBase64 = "";
		МаркаПослеПреобразований = "";
		try
			МаркаВBase64 = СписокТоваров.МаркаВBase64;
		except            
			МаркаВBase64 = "";
		endtry;
		Если ПустоеЗначение(МаркаВBase64) = 1 тогда
			МаркаСырая = ПолучитьСыруюМаркуИзСтрокиЧека(СписокТоваров);
			Если ПустаяСтрока(МаркаСырая) = 0 тогда
				МаркаПослеПреобразований = ПодготовитьМаркуДляККТ(МаркаСырая);
				МаркаВBase64 = КодироватьBase64(МаркаПослеПреобразований);
			иначе
				МаркаВBase64 = "";
			конецЕсли;
		иначе
			МаркаПослеПреобразований = ДекодироватьBase64(МаркаВBase64);
		конецЕсли;		
		Если СокрЛП(МаркаВBase64) = "" тогда
			Продолжить;
		конецЕсли;		
		для сч = 1 по колМарок цикл
			маркаСп = списокМарок.ПолучитьЗначение(сч, );
			Если СокрЛП(маркаСп.Получить("МаркаВBase64")) = МаркаВBase64 тогда
				Если Число(маркаСп.Получить("Принята")) = 1 тогда
					СписокТоваров.МаркаПринятаККТ = 1;
				конецЕсли;
				Если СокрЛП(маркаСп.Получить("UUID")) <> "" тогда
					СписокТоваров.UUID = СокрЛП(маркаСп.Получить("UUID"));
				конецЕсли;
				Если СокрЛП(маркаСп.Получить("time")) <> "" тогда
					СписокТоваров.time = СокрЛП(маркаСп.Получить("time"));
				конецЕсли;
				Если СокрЛП(маркаСп.Получить("Inst")) <> "" тогда
					СписокТоваров.Inst = СокрЛП(маркаСп.Получить("Inst"));
				конецЕсли;
				Если СокрЛП(маркаСп.Получить("Version")) <> "" тогда
					СписокТоваров.Version = СокрЛП(маркаСп.Получить("Version"));
				конецЕсли;				
				Прервать;
			конецЕсли;
		конецЦикла;
	КонецЦикла;
конецПроцедуры //ПеренестиПринятиеМарокККТВТаблицуЧека

функция СформироватьJSONПозицииЧека(СписокТоваров, ПризнВозврата, НомерСекции)

	json = json + """department"": " + Формат(НомерСекции, "Ч15.0") + ", ";
	json = json + """tax"": {""type"": ""none""}, ""paymentObject"": ""commodity"", ""paymentMethod"": ""fullPayment"", ""measurementUnit"": ""piece""";

			МаркаВBase64 = СписокТоваров.МаркаВBase64;

			МаркаВBase64 = "";
		endtry;		
		Если ПустоеЗначение(МаркаВBase64) = 1 тогда

функция СобратьJSONПлатежейLine(СуммаЧека, Получено, ПолученоПоБезналу, ПолученоКарта, ПолученоПоКредиту, ЗачетАванса)
	payments = "";
	первый = 1;
	суммаНал = 0;
	Если ЗачетАванса > 0 тогда
		Если первый = 0 тогда payments = payments + ", "; конецЕсли;
		первый = 0;
		payments = payments + "{""type"": ""prepaid"", ""sum"": " + ЧислоДляJSON(ЗачетАванса) + "}";
	конецЕсли;
	Если ПолученоКарта > 0 тогда
		Если первый = 0 тогда payments = payments + ", "; конецЕсли;
		первый = 0;
		payments = payments + "{""type"": ""prepaid"", ""sum"": " + ЧислоДляJSON(ПолученоКарта) + "}";
	конецЕсли;
	Если ПолученоПоБезналу > 0 тогда
		Если первый = 0 тогда payments = payments + ", "; конецЕсли;
		первый = 0;
		payments = payments + "{""type"": ""electronically"", ""sum"": " + ЧислоДляJSON(ПолученоПоБезналу) + "}";
	конецЕсли;
	Если ПолученоПоКредиту > 0 тогда
		Если первый = 0 тогда payments = payments + ", "; конецЕсли;
		первый = 0;
		payments = payments + "{""type"": ""credit"", ""sum"": " + ЧислоДляJSON(ПолученоПоКредиту) + "}";
	конецЕсли;
	Если Получено > 0 тогда
		суммаНал = Получено;
	иначеЕсли (ПолученоКарта = 0) и (ПолученоПоБезналу = 0) и (ПолученоПоКредиту = 0) и (ЗачетАванса = 0) тогда
		суммаНал = СуммаЧека;
	иначеЕсли (СуммаЧека - ПолученоКарта - ПолученоПоБезналу - ПолученоПоКредиту - ЗачетАванса) > 0 тогда
		суммаНал = СуммаЧека - ПолученоКарта - ПолученоПоБезналу - ПолученоПоКредиту - ЗачетАванса;
	конецЕсли;
	Если суммаНал > 0 тогда
		Если первый = 0 тогда payments = payments + ", "; конецЕсли;
		первый = 0;
		payments = payments + "{""type"": ""cash"", ""sum"": " + ЧислоДляJSON(суммаНал) + "}";
	конецЕсли;
	Если первый = 1 тогда
		payments = "{""type"": ""cash"", ""sum"": " + ЧислоДляJSON(СуммаЧека) + "}";
	конецЕсли;
	Возврат payments;
конецФункции //СобратьJSONПлатежейLine

Функция ДекодироватьBase64(Текст64)
    Перем Алфавит;
    Перем Группа, Символ;
    Перем Значение, Разряд;
    Перем Символ6, Значение6;
    Перем Значение8, Символ8;
    Перем Текст;
    
    // Подготовка алфавита и результата
    Алфавит = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/=";
    Текст = "";
    
    // Получение групп по 4 значащих символа
    Группа = "";
    Для Номер = 1 По СтрДлина(Текст64) Цикл
        Символ = Сред(Текст64, Номер, 1);
        Если Найти(Алфавит, Символ) = 0 Тогда
            Продолжить;
        КонецЕсли;
        Группа = Группа + Символ;
        Если СтрДлина(Группа) = 4 Тогда


            // Получение числового значения из 4-х символов
            Значение = 0;
            Разряд = 3;
            Пока Разряд >= 0 Цикл
                Символ6 = Сред(Группа, 4 - Разряд, 1);
                Если Символ6 = "=" Тогда
                    Значение6 = 0;
                Иначе
                    Значение6 = Найти(Алфавит, Символ6) - 1;
                КонецЕсли;
                Если Разряд = 1 Тогда
                    Значение6 = Значение6 * 64;
                ИначеЕсли Разряд = 2 Тогда
                    Значение6 = Значение6 * 64 * 64;
                ИначеЕсли Разряд = 3 Тогда
                    Значение6 = Значение6 * 64 * 64 * 64;
                КонецЕсли;
                Значение = Значение + Значение6;
                Разряд = Разряд - 1;
            КонецЦикла;


            // Получение 3-х символов из числового значения            
            Разряд = 2;
            Пока Разряд >= 0 Цикл
                Значение8 = Значение;
                Если Разряд = 2 Тогда
                    Значение8 = Цел(Значение8 / 256 / 256);
                    Значение = Значение - Значение8 * 256 * 256;
                ИначеЕсли Разряд = 1 Тогда
                    Значение8 = Цел(Значение8 / 256);
                    Значение = Значение - Значение8 * 256;
                КонецЕсли;
                
                Если Значение8 <>  0 Тогда // [+] Decker
                    // чтобы не добавлялись лишние символы пробела
                    Символ8 = Симв(Значение8);
                    Текст = Текст + Символ8;
                КонецЕсли;
                
                Разряд = Разряд - 1;
            КонецЦикла;
            
            Группа = "";
        КонецЕсли;
    КонецЦикла;
    
    Возврат Текст;    
КонецФункции // ДекодироватьBase64()
