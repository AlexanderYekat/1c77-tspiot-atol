Функция JSONЭкранироватьСтроку(Стр)

		симвв = Сред(Стр, н, 1);
		Если симвв = "\" Тогда
			рез = рез + "\\";
		ИначеЕсли симвв = """" Тогда
			рез = рез + "\""";
		ИначеЕсли симвв = Симв(10) Тогда
			рез = рез + "\n";
		ИначеЕсли симвв = Симв(13) Тогда
			рез = рез + "\r";
		ИначеЕсли симвв = Симв(9) Тогда
			рез = рез + "\t";
		Иначе
			рез = рез + симвв;
		КонецЕсли;
	КонецЦикла;
	Возврат рез;

процедура ОбработатьКомандуККТПриОткрытии(передПараметры)
	команда = СокрЛП(передПараметры.Получить("команда"));
	спРез = СоздатьОбъект("СписокЗначений");
	спРез.Установить("Результат", 0);
	спРез.Установить("кодОшибки", 40);
	спРез.Установить("ОписаниеРезультата", "неизвестная команда ККТ: """ + команда + """");
	Если команда = "доступность" тогда
		спРез = ПроверитьДоступностьПрокси();
	иначеесли команда = "connect" тогда
		спРез = ПодключитьсяКККТЧерезПрокси();		
	иначеесли команда = "info" тогда
		спРез = ПолучитьИННиФНКассыЧерезПрокси();
	иначеесли команда = "connection-status" тогда
		спРез = ПолучитьСтатусПодключенияККТЧерезПрокси();
	иначеесли команда = "print-check" тогда
		спРез = ПечатьЧекаЧерезПрокси(передПараметры.Получить("json"));
	иначеесли команда = "check-mark" тогда
		списокМарок = передПараметры.Получить("СписокМарок");
		асинхронно = передПараметры.Получить("Ассинхронно");
		признВозврата = передПараметры.Получить("ПризнВозврата");
		кассир = передПараметры.Получить("Кассир");
		проверятьРР = передПараметры.Получить("ПроверятьРР");
		ИНН = передПараметры.Получить("ИНН");
		ФН = передПараметры.Получить("ФН");

			асинхронно = 0;
		конецЕсли;

		конецЕсли;
		Если ПустоеЗначение(проверятьРР) = 1 тогда

		конецЕсли;

		спРез = ЗакрытьСменуЧерезПрокси();

		кассир = передПараметры.Получить("cashierName");

        Символ = Сред(Стр, Н, 1);       
        Если Символ = """" Тогда

            ВКавычках = 1 - ВКавычках;
			Если ВКавычках = 1 тогда
				Если ТекущееЗначение <> "" тогда
					ТекущееЗначение = ТекущееЗначение + Символ;
				конецЕсли;				
			конецЕсли;
            Продолжить;
        КонецЕсли;        
        Если (Символ = ",") И (НЕ (ВКавычках = 1)) Тогда
            Результат.ДобавитьЗначение(ТекущееЗначение);
            ТекущееЗначение = "";
            Продолжить;
        КонецЕсли;        
        ТекущееЗначение = ТекущееЗначение + Символ;
    КонецЦикла;

    Результат.ДобавитьЗначение(ТекущееЗначение);

Функция ПреобразоватьСтрокуВУТФ8(Стр)
    Длина=СтрДлина(Стр);
    Итог="";
    Для Н=1 По Длина Цикл
        Знак=Сред(Стр,Н,1);
        Код=КодСимв(Знак);
        Если Код<128 Тогда
            Итог=Итог+Знак;
        Иначе
            Если (Код>=КодСимв("А"))И(Код<=КодСимв("п")) Тогда
                Итог=Итог+Симв(208)+Симв(144+Код-КодСимв("А"));
            ИначеЕсли (Код>=КодСимв("р"))И(Код<=КодСимв("я")) Тогда
                Итог=Итог+Симв(209)+Симв(128+Код-КодСимв("р"));
            ИначеЕсли (Знак="ё") Тогда
                Итог=Итог+Симв(209)+Симв(145);
            ИначеЕсли (Знак="Ё") Тогда
                Итог=Итог+Симв(208)+Симв(129);
            КонецЕсли;
        КонецЕсли;
    КонецЦикла;
    Возврат Итог;
